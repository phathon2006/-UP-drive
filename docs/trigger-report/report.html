รายงาน Database Trigger

ระบบแชร์รถมหาวิทยาลัยพะเยา — UP Drive

รายงานแสดง Trigger ของกลุ่ม (Database Trigger / PostgreSQL) พร้อมผลการทดสอบจริง

1. Trigger คืออะไร

Trigger (ทริกเกอร์) คือชุดคำสั่งที่ฐานข้อมูลเรียกให้ทำงาน อัตโนมัติ เมื่อเกิดเหตุการณ์ (event) กับตาราง เช่น มีการ INSERT, UPDATE หรือ DELETE โดยไม่ต้องให้โปรแกรมฝั่งแอปสั่งเอง ทำให้กติกาของระบบถูกบังคับใช้ที่ฐานข้อมูลเสมอ แม้มีโปรแกรมหรือคน เข้าถึงข้อมูลหลายทาง

ใน PostgreSQL/ Supabase ประกอบด้วย 2 ส่วน:

  • Trigger Function — ฟังก์ชันภาษา PL/pgSQL ที่ทำงานจริงและ RETURN NEW
  • CREATE TRIGGER — ผูกฟังก์ชันกับตาราง ระบุว่าให้ทำงาน ก่อน (BEFORE) หรือ หลัง (AFTER) การแก้ไข

2. ทำไมกลุ่มเราจึงเลือกใช้ Trigger

ระบบ UP Drive เดิมตั้งค่าเวลาและตรวจสอบกติกาจาก โค้ดฝั่งเซิร์ฟเวอร์ (Node.js) ซึ่งมีข้อเสียคือ ถ้ามีการแก้ข้อมูลจากช่องทางอื่น (เช่น SQL Editor หรือ API ตรง) กติกาจะถูกข้ามได้ กลุ่มเราจึงย้ายกติกาสำคัญ 3 ข้อไปเป็น Trigger เพื่อให้ฐานข้อมูลคุมเอง:

  • ตั้งเวลาอัตโนมัติ — เมื่อกลุ่มเปลี่ยนสถานะ ให้บันทึกเวลา confirmed_at / accepted_at / completed_at ให้เอง
  • บังคับกติกาความถูกต้อง — ห้ามตั้งสถานะ accepted ถ้ายังไม่มีคนขับ
  • อัปเดตตำแหน่งล่าสุด — เมื่อพิกัดคนขับถูกแก้ ให้ updated_at เป็นเวลาปัจจุบันเสมอ

3. โค้ด Trigger ของกลุ่ม

ด้านล่างคือสคริปต์ Trigger ทั้งหมด (ไฟล์ trigger.sql) ที่รันบน PostgreSQL ของ Supabase

-- =========================================================
-- UP DRIVE V2 - DATABASE TRIGGERS
-- กลุ่ม: UP Drive (ระบบแชร์รถ มหาวิทยาลัยพะเยา)
-- ไฟล์นี้รวม TRIGGER ทั้งหมดของระบบ
-- =========================================================

-- ---------------------------------------------------------
-- TRIGGER 1: ตั้งเวลาอัตโนมัติเมื่อสถานะของกลุ่มเปลี่ยน
--   forming -> sent_to_driver  => confirmed_at = now()
--   -> accepted                => accepted_at  = now()
--   -> completed               => completed_at = now()
-- ---------------------------------------------------------
create or replace function public.sync_group_status_timestamps()
returns trigger
language plpgsql
as $$
begin
  -- ทำงานเฉพาะตอน "สถานะเปลี่ยนจริง" (idempotent)
  if new.status is distinct from old.status then

    if new.status = 'sent_to_driver'
       and new.confirmed_at is null then
      new.confirmed_at := now();
    end if;

    if new.status = 'accepted'
       and new.accepted_at is null then
      new.accepted_at := now();
    end if;

    if new.status = 'completed'
       and new.completed_at is null then
      new.completed_at := now();
    end if;

  end if;

  return new;
end;
$$;

drop trigger if exists trg_ride_groups_status_timestamps
on public.ride_groups;

create trigger trg_ride_groups_status_timestamps
before update on public.ride_groups
for each row
execute function public.sync_group_status_timestamps();


-- ---------------------------------------------------------
-- TRIGGER 2: บังคับกติกาความถูกต้องของข้อมูล
--   ห้ามตั้งสถานะ accepted ถ้ายังไม่มีคนขับ (driver_id is null)
-- ---------------------------------------------------------
create or replace function public.enforce_group_driver_rules()
returns trigger
language plpgsql
as $$
begin
  if new.status = 'accepted' and new.driver_id is null then
    raise exception
      'ride_groups: ตั้งสถานะ accepted ไม่ได้ถ้าไม่มีคนขับ (driver_id)';
  end if;

  return new;
end;
$$;

drop trigger if exists trg_ride_groups_driver_rules
on public.ride_groups;

create trigger trg_ride_groups_driver_rules
before update on public.ride_groups
for each row
execute function public.enforce_group_driver_rules();


-- ---------------------------------------------------------
-- TRIGGER 3: อัปเดต updated_at ของตำแหน่งคนขับให้เป็นปัจจุบันเสมอ
-- ---------------------------------------------------------
create or replace function public.touch_driver_location()
returns trigger
language plpgsql
as $$
begin
  new.updated_at := now();
  return new;
end;
$$;

drop trigger if exists trg_driver_locations_touch
on public.driver_locations;

create trigger trg_driver_locations_touch
before update on public.driver_locations
for each row
execute function public.touch_driver_location();

ภาพที่ 1 — โค้ด Trigger ทั้ง 3 ตัวของกลุ่ม

ภาพที่ 2 — ลำดับการเปลี่ยนสถานะ และเวลาที่ Trigger เติมให้ (ช่องสีเขียว)

5. ผลการทดสอบ — Trigger บังคับกติกาความถูกต้อง

เมื่อพยายามตั้งสถานะเป็น accepted ให้กลุ่มที่ยังไม่มีคนขับ ฐานข้อมูลปฏิเสธคำสั่งทันที พร้อมข้อความแจ้งเตือน ข้อมูลจึงไม่ถูกบันทึก (สถานะคงเป็น forming)

6พยายามตั้ง accepted โดยไม่มีคนขับ => ถูก trigger บล็อก
update public.ride_groups
set status = 'accepted'
where id = 'bbbbbbbb-2222-2222-2222-222222222222';

8. สรุป

  • กลุ่มเราเพิ่ม Database Trigger จำนวน 3 ตัว ให้กับตาราง ride_groups และ driver_locations
  • Trigger ช่วยให้ฐานข้อมูล เติมเวลาและบังคับกติกาเองอัตโนมัติ ลดการพึ่งพาโค้ดฝั่งเซิร์ฟเวอร์
  • ทดสอบจริงแล้วพบว่า: เวลาถูกเติมถูกต้อง, ค่าเดิมไม่ถูกเขียนทับ, ข้อมูลผิดกติกาถูกบล็อก, และ updated_at อัปเดตเอง
  • ผลทดสอบรันบน PostgreSQL 16 (จำลองด้วย PGlite/WASM) ด้วย SQL ชุดเดียวกับ Supabase

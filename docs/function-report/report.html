รายงาน SQL Function

รายงานแสดง SQL Function ของกลุ่ม พร้อมผลการเรียกใช้งานจริง

1. SQL Function คืออะไร

SQL Function (ฟังก์ชัน) คือชุดคำสั่ง SQL ที่ถูกเก็บไว้ในฐานข้อมูล ตั้งชื่อไว้ แล้ว เรียกใช้ซ้ำได้ ด้วยชื่อฟังก์ชัน พร้อมส่งค่า (parameter) เข้าไปได้ เหมือนฟังก์ชันในภาษาโปรแกรม แต่ทำงานอยู่ที่ฐานข้อมูล

ข้อดีคือ เขียนตรรกะครั้งเดียว แล้วเรียกใช้ได้จากทุกที่ — จาก SQL Editor, จากฟังก์ชันอื่น หรือจากแอปพลิเคชันผ่าน supabase.rpc()

2. ประเภทของฟังก์ชันที่ใช้ในงานนี้

  • Scalar function — คืนค่าเดี่ยว ๆ 1 ค่า เช่น haversine_km() คืนตัวเลขระยะทาง
  • Table function — คืนผลเป็นตาราง เรียกด้วย select * from fn(...) เช่น find_candidate_groups()
  • เขียนด้วยภาษา SQL (language sql) จึงอ่านง่ายและให้ฐานข้อมูล optimize เอง

3. ทำไมกลุ่มเราจึงเลือกเขียน Function

เดิมตรรกะ "คำนวณระยะทาง" และ "จับคู่กลุ่มที่ใกล้ที่สุด" อยู่ที่โค้ดฝั่งเซิร์ฟเวอร์ (Node.js) กลุ่มเราย้ายมาเป็น SQL Function เพื่อให้ฐานข้อมูลคำนวณได้เอง ใช้ซ้ำได้ทุกที่ และทดสอบได้ทันทีใน SQL Editor

4. โค้ด SQL Function ของกลุ่ม

สคริปต์ทั้งหมดอยู่ในไฟล์ functions.sql — ประกอบด้วย 2 scalar function และ 1 table function

-- =========================================================
-- UP DRIVE V2 - SQL FUNCTIONS
-- ระบบแชร์รถ มหาวิทยาลัยพะเยา (กลุ่ม UP Drive)
-- รันบน PostgreSQL / Supabase
-- =========================================================

-- ---------------------------------------------------------
-- FUNCTION 1: haversine_km  (scalar function, ภาษา SQL)
--   คำนวณระยะทางระหว่างพิกัด 2 จุด หน่วยเป็นกิโลเมตร
--   ใช้สูตร Haversine เหมือนที่ฝั่งแอปเคยคำนวณเอง
-- ---------------------------------------------------------
create or replace function public.haversine_km(
  lat1 double precision,
  lng1 double precision,
  lat2 double precision,
  lng2 double precision
)
returns double precision
language sql
immutable
as $$
  select round(
    (
      6371 * 2 * asin(
        sqrt(
          power(sin(radians(lat2 - lat1) / 2), 2) +
          cos(radians(lat1)) * cos(radians(lat2)) *
          power(sin(radians(lng2 - lng1) / 2), 2)
        )
      )
    )::numeric,
    3
  )::double precision;
$$;


-- ---------------------------------------------------------
-- FUNCTION 2: group_passenger_total  (scalar function, ภาษา SQL)
--   นับจำนวนผู้โดยสารรวมของกลุ่ม (ผลรวม party_size ของสมาชิก)
--   ถ้ากลุ่มยังไม่มีสมาชิก จะคืนค่า 0
-- ---------------------------------------------------------
create or replace function public.group_passenger_total(
  p_group_id uuid
)
returns integer
language sql
stable
as $$
  select coalesce(sum(m.party_size), 0)::integer
  from public.ride_group_members m
  where m.group_id = p_group_id;
$$;


-- ---------------------------------------------------------
-- FUNCTION 3: find_candidate_groups  (table function, ภาษา SQL)
--   หา "กลุ่มที่กำลังรวมคน (forming)" ที่อยู่ใกล้คำขอเดินทางใหม่
--   แล้วคืนผลเป็นตาราง พร้อมคะแนน (ยิ่งน้อย = ยิ่งใกล้)
--   เกณฑ์เริ่มต้น: จุดรับ <= 4 km และปลายทาง <= 6 km
-- ---------------------------------------------------------
create or replace function public.find_candidate_groups(
  p_origin_lat double precision,
  p_origin_lng double precision,
  p_dest_lat double precision,
  p_dest_lng double precision,
  p_max_pickup_km double precision default 4,
  p_max_dest_km double precision default 6
)
returns table (
  group_id uuid,
  leader_id uuid,
  target_passengers integer,
  current_passengers integer,
  pickup_km double precision,
  destination_km double precision,
  score double precision
)
language sql
stable
as $$
  select
    g.id,
    g.leader_id,
    g.target_passengers,
    public.group_passenger_total(g.id),
    public.haversine_km(p_origin_lat, p_origin_lng, r.origin_lat, r.origin_lng),
    public.haversine_km(p_dest_lat, p_dest_lng, r.destination_lat, r.destination_lng),
    public.haversine_km(p_origin_lat, p_origin_lng, r.origin_lat, r.origin_lng)
      + public.haversine_km(p_dest_lat, p_dest_lng, r.destination_lat, r.destination_lng)
  from public.ride_groups g
  join public.ride_requests r on r.group_id = g.id
  where g.status = 'forming'
    and public.haversine_km(p_origin_lat, p_origin_lng, r.origin_lat, r.origin_lng)
        <= p_max_pickup_km
    and public.haversine_km(p_dest_lat, p_dest_lng, r.destination_lat, r.destination_lng)
        <= p_max_dest_km
  order by 7 asc;
$$;

ภาพที่ 1 — โค้ด SQL Function ทั้ง 3 ตัวของกลุ่ม

5. ผลการทดสอบ — Scalar Functions

2เรียก FUNCTION 1: haversine_km() — วัดระยะทางจากจุดรับใหม่ไปยังแต่ละกลุ่ม
select * from (values
  ('จุดรับใหม่ -> กลุ่ม A', public.haversine_km(19.0295, 99.8955, 19.0302, 99.8962)),
  ('จุดรับใหม่ -> กลุ่ม C', public.haversine_km(19.0295, 99.8955, 19.0320, 99.8980)),
  ('จุดรับใหม่ -> กลุ่ม B', public.haversine_km(19.0295, 99.8955, 19.0600, 99.9300))
) as t(คู่พิกัด, ระยะทาง_km);

8. สรุป

  • กลุ่มเราเพิ่ม SQL Function จำนวน 3 ตัว (2 scalar + 1 table function) เขียนด้วยภาษา SQL
  • haversine_km() ใช้คำนวณระยะทางจริง — ทดสอบแล้วให้ผล 0.107 / 0.383 / 4.965 km ตรงตามพิกัด
  • group_passenger_total() นับผู้โดยสารรวมของกลุ่มได้ถูกต้อง และคืน 0 เมื่อกลุ่มว่าง
  • find_candidate_groups() คืนกลุ่มที่ใกล้ที่สุดพร้อมคะแนน และกรองกลุ่มที่ไกลเกินระยะได้จริง
